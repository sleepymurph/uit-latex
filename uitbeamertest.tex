\begin{frame}{Angle Test}
    \begin{itemize}
        \item The split can be drawn with different parameters\\
            \tikz{\uitsplitbackground{0,0}{4em, 3em}}
            \tikz{\uitsplitbackground[
                    split pos = .25,
                    split right/.append style = {color=UitRed, opacity=.4},
                ]{0,0}{4em, 3em}}
            \tikz{\uitsplitbackground[
                    opacity=.6,
                    line width = 1em,
                ]{0,0}{4em, 3em}}
            \tikz{\uitsplitbackground[split rays=true]{0,0}{4em, 3em}}

        \item The split can be placed with transparency, like in
            \href{https://uit.no/ansatte/grafiskprofil\#innhold_631815}{UiT's examples}\\

            \begin{tikzpicture}
                \node[inner sep=0, outer sep=0] (bg)
                    {\includegraphics[height=5em]{example-art/csiro-five-antennas-bg.jpg}};
                \uitsplitbackground[
                    split pos = .55,
                    split left/.style = {},
                    split right/.append style = {color=UitRed, opacity=.4},
                    ]{bg.south west}{bg.north east}
            \end{tikzpicture}
            \begin{tikzpicture}
                \node[inner sep=0, outer sep=0] (bg)
